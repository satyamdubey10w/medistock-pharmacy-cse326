function openSupplierModal(s=null){if(currentUser?.role!=='admin'){toast('Only admin can manage suppliers.');return}editingSupplier=s?.id||null;let v=s||{name:'',contact:'',phone:'',email:'',address:''};openModal(s?'Edit supplier':'Add supplier',`<form id="supplierForm"><div class="form-grid"><div class="field full"><label>Supplier / company name *</label><input name="name" required value="${esc(v.name)}"></div><div class="field"><label>Contact person</label><input name="contact" value="${esc(v.contact)}"></div><div class="field"><label>Phone</label><input name="phone" value="${esc(v.phone)}"></div><div class="field"><label>Email</label><input name="email" type="email" value="${esc(v.email)}"></div><div class="field"><label>Address</label><input name="address" value="${esc(v.address)}"></div></div><div class="form-actions"><button type="button" class="outline" onclick="closeModal()">Cancel</button><button class="primary">Save supplier</button></div></form>`);$('supplierForm').onsubmit=e=>{e.preventDefault();let d=Object.fromEntries(new FormData(e.target));if(editingSupplier)suppliers=suppliers.map(x=>x.id===editingSupplier?{...x,...d}:x);else suppliers.push({id:uid('SUP'),...d});save();closeModal();renderAll();toast('Supplier saved.')}}
function editSupplier(id){let s=suppliers.find(x=>x.id===id);if(s)openSupplierModal(s)}
function deleteSupplier(id){if(currentUser?.role!=='admin')return;if(confirm('Delete this supplier record?')){suppliers=suppliers.filter(s=>s.id!==id);save();renderAll();toast('Supplier deleted.')}}
function renderPurchases(){$('purchaseRows').innerHTML=purchases.map(p=>`<tr><td><b>${esc(p.id)}</b></td><td>${esc(p.date)}</td><td>${esc(p.supplier)}</td><td>${esc(p.medicine)}</td><td>${p.qty}</td><td>${money(p.cost)}</td><td><b>${money(p.qty*p.cost)}</b></td></tr>`).join('')||'<tr><td colspan="7" class="empty">No purchase records yet.</td></tr>'}
function openPurchaseModal(){if(currentUser?.role!=='admin'){toast('Only admin can record purchases.');return}openModal('Record a purchase',`<form id="purchaseForm"><div class="form-grid"><div class="field full"><label>Medicine *</label><select name="medicineId" required>${medicines.map(m=>`<option value="${m.id}">${esc(m.name)} (${m.stock} in stock)</option>`).join('')}</select></div><div class="field"><label>Supplier *</label><select name="supplier" required>${suppliers.map(s=>`<option>${esc(s.name)}</option>`).join('')}</select></div><div class="field"><label>Quantity received *</label><input name="qty" type="number" min="1" required value="10"></div><div class="field"><label>Purchase cost / unit (₹) *</label><input name="cost" type="number" min="0" step=".01" required value="1"></div><div class="field"><label>Purchase date</label><input name="date" type="date" required value="${today()}"></div></div><div class="form-actions"><button type="button" class="outline" onclick="closeModal()">Cancel</button><button class="primary">Save purchase & update stock</button></div></form>`);$('purchaseForm').onsubmit=e=>{e.preventDefault();let d=Object.fromEntries(new FormData(e.target)),m=medicines.find(x=>x.id===d.medicineId);d.qty=+d.qty;d.cost=+d.cost;if(!m)return;m.stock+=d.qty;m.cost=d.cost;m.supplier=d.supplier;purchases.unshift({id:uid('PUR'),date:d.date,supplier:d.supplier,medicine:m.name,medicineId:m.id,qty:d.qty,cost:d.cost});save();closeModal();renderAll();toast('Purchase recorded and stock increased.')}}
function renderPrescriptions(){let q=($('prescriptionSearch')?.value||'').toLowerCase();let arr=prescriptions.filter(p=>(p.id+' '+p.patient+' '+p.prescriber+' '+p.notes).toLowerCase().includes(q));$('prescriptionRows').innerHTML=arr.map(p=>`<tr><td><b>${esc(p.id)}</b></td><td>${esc(p.date)}</td><td>${esc(p.patient)}</td><td>${esc(p.prescriber)}</td><td>${esc(p.notes)}</td><td><button class="danger small" onclick="deletePrescription('${p.id}')">Delete</button></td></tr>`).join('')||'<tr><td colspan="6" class="empty">No prescription references recorded.</td></tr>'}
$('prescriptionSearch').addEventListener('input',renderPrescriptions);
function openPrescriptionModal(){openModal('Add prescription reference',`<form id="rxForm"><div class="form-grid"><div class="field"><label>Record date *</label><input name="date" type="date" required value="${today()}"></div><div class="field"><label>Patient label *</label><input name="patient" required placeholder="Demo patient"></div><div class="field"><label>Prescriber *</label><input name="prescriber" required placeholder="Doctor / clinic"></div><div class="field full"><label>Medicine notes / reference *</label><textarea name="notes" required rows="3" placeholder="Reference only — avoid real patient details"></textarea></div></div><div class="form-actions"><button type="button" class="outline" onclick="closeModal()">Cancel</button><button class="primary">Save record</button></div></form>`);$('rxForm').onsubmit=e=>{e.preventDefault();let d=Object.fromEntries(new FormData(e.target));prescriptions.unshift({id:uid('RX'),...d});save();closeModal();renderAll();toast('Prescription reference saved.')}}
function deletePrescription(id){if(confirm('Delete this reference record?')){prescriptions=prescriptions.filter(p=>p.id!==id);save();renderAll();toast('Record deleted.')}}
function renderReports(){$('reportInventory').textContent=money(medicines.reduce((a,m)=>a+m.stock*m.price,0));$('reportSales').textContent=money(sales.reduce((a,s)=>a+s.total,0));$('reportPurchases').textContent=money(purchases.reduce((a,p)=>a+p.qty*p.cost,0));$('reportRx').textContent=prescriptions.length;let days=[];for(let i=6;i>=0;i--){let d=new Date();d.setDate(d.getDate()-i);let key=d.toISOString().slice(0,10),val=sales.filter(s=>s.date.slice(0,10)===key).reduce((a,s)=>a+s.total,0);days.push({key,label:key.slice(5),val})}let max=Math.max(1,...days.map(d=>d.val));$('salesChart').innerHTML=days.map(d=>`<div class="barrow"><span>${d.label}</span><div class="bar"><span style="width:${d.val/max*100}%"></span></div><b>${money(d.val)}</b></div>`).join('');let groups={'In stock':0,'Low stock':0,'Expiring soon':0,'Expired':0};medicines.forEach(m=>groups[stockStatus(m)[0]]++);let total=Math.max(1,medicines.length);$('inventoryHealth').innerHTML=Object.entries(groups).map(([k,v])=>`<div class="barrow"><span>${k}</span><div class="bar"><span style="width:${v/total*100}%;background:${k==='Expired'||k==='Low stock'?'#e05b5b':k==='Expiring soon'?'#e4a32e':''}"></span></div><b>${v}</b></div>`).join('')}
function csvDownload(filename,rows){let csv=rows.map(r=>r.map(v=>'"'+String(v??'').replace(/"/g,'""')+'"').join(',')).join('\r\n');let blob=new Blob(['\ufeff'+csv],{type:'text/csv;charset=utf-8;'}),url=URL.createObjectURL(blob),a=document.createElement('a');a.href=url;a.download=filename;a.click();URL.revokeObjectURL(url)}
function exportMedicines(){csvDownload('medistock-inventory.csv',[['ID','Name','Category','Batch','Stock','Minimum','Price','Cost','Expiry','Supplier','Status'],...medicines.map(m=>[m.id,m.name,m.category,m.batch,m.stock,m.min,m.price,m.cost,m.expiry,m.supplier,stockStatus(m)[0]])])}
function exportSales(){csvDownload('medistock-sales.csv',[['Invoice','Customer','Date','Items','Subtotal','Discount','Total'],...sales.map(s=>[s.id,s.customer,s.date,s.items.map(i=>i.name+' x'+i.qty).join('; '),s.subtotal,s.discount,s.total])])}
function exportPurchases(){csvDownload('medistock-purchases.csv',[['Purchase ID','Date','Supplier','Medicine','Quantity','Cost per unit','Total'],...purchases.map(p=>[p.id,p.date,p.supplier,p.medicine,p.qty,p.cost,p.qty*p.cost])])}
function exportAll(){csvDownload('medistock-inventory-export.csv',[['DATASET','ID','Name / Detail','Date','Quantity','Amount','Extra'],...medicines.map(m=>['Medicine',m.id,m.name,m.expiry,m.stock,m.price,stockStatus(m)[0]]),...sales.map(s=>['Sale',s.id,s.customer,s.date,s.items.reduce((a,i)=>a+i.qty,0),s.total,s.items.map(i=>i.name).join('; ')]),...purchases.map(p=>['Purchase',p.id,p.supplier,p.date,p.qty,p.qty*p.cost,p.medicine]),...prescriptions.map(p=>['Prescription',p.id,p.patient,p.date,'', '',p.notes])])}
